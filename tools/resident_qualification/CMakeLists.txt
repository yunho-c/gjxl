cmake_minimum_required(VERSION 3.24)
project(gjxl_resident_qualification LANGUAGES C CXX OBJCXX)
if(NOT GJXL_QUALIFICATION_SOURCE OR NOT GJXL_QUALIFICATION_CAPABILITY MATCHES "^(legacy|current)$")
  message(FATAL_ERROR "Supply a source tree and legacy/current capability")
endif()
enable_testing()
add_subdirectory("${GJXL_QUALIFICATION_SOURCE}" codec)
add_executable(gjxl_qualification_driver driver.cpp)
target_compile_features(gjxl_qualification_driver PRIVATE cxx_std_20)
target_link_libraries(gjxl_qualification_driver PRIVATE gjxl_codestream gjxl_pfm_io gjxl_metal)
if(GJXL_QUALIFICATION_CAPABILITY STREQUAL "current")
  target_compile_definitions(gjxl_qualification_driver PRIVATE GJXL_FINAL_SCHEDULER)
endif()
add_executable(gjxl_qualification_embedding embedding.cpp)
target_link_libraries(gjxl_qualification_embedding PRIVATE gjxl_metal)
